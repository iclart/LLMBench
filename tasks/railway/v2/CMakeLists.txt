function(add_v2_driver name)
  add_executable(railv2_run_${name} tools/run.cpp)
  target_link_libraries(railv2_run_${name} PRIVATE railv2_${name} railcommon)
  if(MINGW)
    target_link_options(railv2_run_${name} PRIVATE -static)
  endif()
  add_executable(railv2_tests_${name} tests/engine_tests.cpp)
  target_link_libraries(railv2_tests_${name} PRIVATE railv2_${name})
  if(MINGW)
    target_link_options(railv2_tests_${name} PRIVATE -static)
  endif()
  add_test(NAME railv2_engine_${name} COMMAND railv2_tests_${name})
endfunction()
