cmake_minimum_required(VERSION 3.20)
project(LLMBench LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

# ThreadSanitizer build: any data race in an engine fails its railbench run.
option(RAIL_TSAN "Build everything with -fsanitize=thread" OFF)
option(RAIL_ASAN_UBSAN "Build everything with address and undefined behavior sanitizers" OFF)
if(RAIL_TSAN AND RAIL_ASAN_UBSAN)
  message(FATAL_ERROR "TSan and ASan/UBSan require separate build trees")
endif()
if((RAIL_TSAN OR RAIL_ASAN_UBSAN) AND WIN32)
  message(FATAL_ERROR "The supported sanitizer validation requires a Linux toolchain")
endif()
if(RAIL_TSAN)
  add_compile_options(-fsanitize=thread -g)
  add_link_options(-fsanitize=thread)
endif()
if(RAIL_ASAN_UBSAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()

enable_testing()
add_subdirectory(tasks/railway)
